"""
LESSON 42: The retrieval pipeline -- turning "search" into one clean function


Lessons 38-41 built every PIECE of this by hand, one at a time:
    - chunk_by_size()    -- split a document into pieces (lesson 38)
    - get_embedding()    -- turn text into a vector (lesson 33)
    - embed_chunks()     -- pair text + vector together (lesson 39)
    - Chroma collection  -- store + search those pairs (lesson 41)


RETRIEVAL is just the name for the step where, given a user's QUESTION, you:
    1. Embed the question (turn it into a vector, same as any other text)
    2. Search your vector store for the closest chunks
    3. Return the chunks' TEXT (not the vectors -- an LLM reads text, not
       floats)


You've already done this exact sequence three separate times (once by hand
with cosine_similarity in lesson 34, once with FAISS in lesson 40, once
with Chroma in lesson 41). The only thing new today is wrapping it into
ONE function -- retrieve() -- so the rest of your program (and eventually
the LLM call in lesson 44/45) can just ask a question and get back clean
text, without caring how the search happens underneath.


This is the exact shape every real RAG system uses:
    question -> retrieve() -> context chunks -> hand to LLM as context
"""


import os

load_dotenv(override=True)

    "Authorization": f"Bearer {voyage_api_key}",

}




def get_embedding(text, input_type="document"):
    """Same helper from lessons 33/34/39/40/41."""
    body = {
        "input": [text],
        "model": "voyage-4-lite",
        "input_type": input_type,
    }
    time.sleep(21)  # free-tier accounts are capped at 3 requests/minute
    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["data"][0]["embedding"]




def chunk_by_size(text, chunk_size, overlap=0):
    """Same helper from lesson 38."""
    if overlap >= chunk_size:
        raise ValueError("overlap must be smaller than chunk_size")
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap
    return chunks




def build_collection(document, collection_name, chunk_size=150, overlap=30):
    """Chunks a document, embeds every chunk, and loads it into a fresh
    Chroma collection. Same pattern as lesson 41's demo, just wrapped into
    one function so we can call it once and move on to the new part."""
    chunks = chunk_by_size(document, chunk_size=chunk_size, overlap=overlap)
    embeddings = [get_embedding(chunk) for chunk in chunks]
    ids = [f"{collection_name}_{i}" for i in range(len(chunks))]


    client = chromadb.Client()
    collection = client.create_collection(name=collection_name)
    collection.add(ids=ids, documents=chunks, embeddings=embeddings)
    return collection




def retrieve(query, collection, k=3):
    """THE NEW IDEA. Given a plain-English question and a Chroma
    collection, this does the full retrieval sequence and hands back just
    the TEXT of the top-k matching chunks -- nothing else. This is the
    function an LLM call will eventually reach for in lesson 44/45."""
    query_embedding = get_embedding(query, input_type="query")
    results = collection.query(query_embeddings=[query_embedding], n_results=k)
    return results["documents"][0]

# --- BONUS: METADATA -- the one blueprint concept you haven't used yet ---
# Every collection.add() call above passed ids, documents, and embeddings --
# never metadata. Metadata is just a dict of EXTRA facts about each chunk
# (which document it came from, a page number, a category, whatever you
# want) that rides along with it in Chroma. The real payoff: you can FILTER
# on metadata BEFORE the similarity search even runs -- e.g. "only search
# chunks where source == 'basketball_doc'" -- something plain vector search
# alone can't do. That's why lesson 41's results dict always showed
# 'metadatas': [[None]] -- you never gave it any.


def build_collection_with_metadata(document, collection_name, source_name,
                                    chunk_size=150, overlap=30):
    """Same as build_collection(), but attaches a metadata dict to every
    chunk: which document it came from, and its position in that document."""
    chunks = chunk_by_size(document, chunk_size=chunk_size, overlap=overlap)
    embeddings = [get_embedding(chunk) for chunk in chunks]
    ids = [f"{collection_name}_{i}" for i in range(len(chunks))]
    metadatas = [{"source": source_name, "chunk_index": i} for i in range(len(chunks))]


    client = chromadb.Client()
    collection = client.create_collection(name=collection_name)
    collection.add(
        ids=ids,
        documents=chunks,
        embeddings=embeddings,
        metadatas=metadatas,  # <-- the new piece
    )
    return collection

"""
LESSON 43: Reranking -- a second, smarter pass over your search results


Every retrieval you've done so far (lessons 34, 40, 41, 42) works the same
way: embed the query, embed the documents, compare vectors with math
(cosine similarity / L2 distance), and return whichever chunks are
numerically closest. That's FAST and CHEAP, but it has a real limitation --
it compares the query and each document SEPARATELY, as two vectors that
never actually "look at" each other. Two sentences can end up with
similar-looking vectors even when one is a much better answer to the
question than the other.


A RERANKER fixes this by being slower and smarter, on purpose, but only
for a SHORTLIST. The real pipeline looks like:


    1. Cheap vector search (retrieve() from lesson 42) -- pull back a
       generous shortlist, say the top 5-10 candidates, fast, over your
       whole collection (could be millions of chunks).
    2. Reranker -- a different kind of model (a "cross-encoder") that
       reads the QUERY and EACH CANDIDATE DOCUMENT TOGETHER, side by side,
       and outputs a fresh relevance score per pair. Much more accurate,
       but too slow/expensive to run over an entire collection -- that's
       why it only ever touches the small shortlist step 1 already
       narrowed things down to.
    3. Keep only the reranker's own top-k -- these become the chunks you
       actually hand to the LLM as context.


This is exactly the "chunk -> embed -> store -> retrieve -> RERANK -> LLM"
order from lesson 38's Q&A. Cheap search narrows a huge pile down to a
handful; the reranker spends its extra accuracy only on that handful.


We're using Voyage's real reranker API (same account as your embeddings):
    POST https://api.voyageai.com/v1/rerank
    body: {"query": ..., "documents": [...], "model": "rerank-2.5-lite",
           "top_k": ..., "return_documents": true}
    response: {"data": [{"index": ..., "relevance_score": ..., "document": ...}, ...]}
    (already sorted by the API, highest relevance_score first)
"""

def rerank(query, documents, top_k=3, model="rerank-2.5-lite"):
    """THE NEW IDEA. Sends the query and a SHORTLIST of candidate documents
    to Voyage's reranker. Unlike get_embedding(), which scores the query and
    a document separately, the reranker looks at each (query, document) PAIR
    together and returns a fresh, more accurate relevance_score per pair --
    already sorted, best match first."""
    body = {
        "query": query,
        "documents": documents,
        "model": model,
        "top_k": top_k,
        "return_documents": True,
    }
    time.sleep(21)  # same free-tier rate limit applies to this endpoint
    response = requests.post(rerank_url, headers=headers, json=body)
    data = response.json()
    return data["data"]  # list of {"index", "relevance_score", "document"}

"""
LESSON 44: RAG evaluation -- how do you know your pipeline actually works?


Lessons 38-43 built a full RAG pipeline: chunk -> embed -> store -> retrieve
-> rerank. But how do you know it's actually GOOD? "It printed something
that looked relevant" isn't a real answer once you're building something
real. RAG evaluation has TWO separate questions, and they need TWO
separate checks:


    1. RETRIEVAL quality: did the pipeline pull back the RIGHT chunk(s) for
       a given question? This has nothing to do with the LLM yet -- it's
       purely "did retrieve()/rerank() find the right needle in the
       haystack." You can measure this with plain code, no AI judgment
       needed, IF you know the right answer ahead of time (a "ground
       truth" test set).


    2. ANSWER quality: once you hand the retrieved chunks to an LLM and it
       writes an answer, is that answer actually TRUE to what the context
       said? An LLM can retrieve the perfect chunk and still write an
       answer that drifts away from it ("hallucination") -- so this needs
       a different check than retrieval quality does.


--- PART 1: Retrieval evaluation (hit rate @ k) ---
You build a small set of TEST CASES -- questions where you already know
which chunk SHOULD answer them (because you wrote the document yourself).
Then you run retrieve() for each question and check: is the expected
chunk's ID actually somewhere in the top-k results? The fraction that hit
is your "hit rate @ k" -- a real, standard retrieval metric.


--- PART 2: Answer evaluation (LLM-as-judge) ---
You generate a real answer with Claude, using the retrieved chunks as
context (this is the "G" in RAG -- generation). Then you make a SECOND
Claude call whose only job is to grade the first answer: does it stay
faithful to the provided context, or does it say things the context never
supported? This is called "LLM-as-judge" -- using a model to grade another
model's output against structured, checkable criteria (same
ask_for_json-style pattern from lesson 32).
"""

def evaluate_retrieval(test_cases, collection, k=3):
    """test_cases is a list of {"question": ..., "expected_id": ...} dicts.
    For each one, retrieves top-k and checks whether the expected chunk's
    ID actually showed up. Prints each result, returns the overall hit rate."""
    hits = 0
    for case in test_cases:
        results = retrieve_with_ids(case["question"], collection, k=k)
        retrieved_ids = [chunk_id for chunk_id, _ in results]
        hit = case["expected_id"] in retrieved_ids
        hits += int(hit)
        print(f"Q: {case['question']}")
        print(f"  expected: {case['expected_id']}")
        print(f"  retrieved: {retrieved_ids}")
        print(f"  HIT" if hit else "  MISS")
        print()
    return hits / len(test_cases)

def chunk_by_size(text, chunk_size, overlap=0):
    """Same helper from lesson 38."""
    if overlap >= chunk_size:
        raise ValueError("overlap must be smaller than chunk_size")
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap
    return chunks




def build_collection(document, collection_name, chunk_size=150, overlap=30):
    """Same helper from lesson 42. Call this ONCE per document."""
    chunks = chunk_by_size(document, chunk_size=chunk_size, overlap=overlap)

    ids = [f"{collection_name}_{i}" for i in range(len(chunks))]


    client = chromadb.Client()
    collection = client.create_collection(name=collection_name)
    collection.add(ids=ids, documents=chunks, embeddings=embeddings)
    return collection

LESSON 46 CAPSTONE: Personal knowledge assistant -- multiple documents,
one search engine, real citations.


This is the final Phase 6 capstone. Lesson 45 built a Q&A bot over ONE
document. This one handles MANY documents at once, and adds the one piece
that's been missing since lesson 42's metadata bonus: knowing WHICH
document an answer actually came from. A real personal knowledge
assistant isn't useful if it can only ever look at one file -- and it
isn't trustworthy if it can't tell you its sources.


The new idea is METADATA-AWARE retrieval:
    - Every chunk, from every document, gets tagged with its source
      document's name when it's added to the collection (lesson 42's
      metadata bonus).
    - When you retrieve a shortlist, you pull back the metadata ALONGSIDE
      the text (not just the text, like lesson 42's plain retrieve()).
    - After reranking narrows the shortlist down to the real top matches,
      you look each surviving chunk's TEXT back up in a text->source
      dictionary to recover which document it came from.
    - The final answer gets built WITH citations: "[python_basics] ..." --
      so you can always trace an answer back to its source, the same way
      a real research assistant would.


Pipeline, one more time, now with citations:
    documents (dict of {source_name: text})
        -> build_knowledge_base() [chunk + embed + tag metadata + store, ONCE]
    question
        -> retrieve_with_metadata() [cheap shortlist, WITH sources attached]
        -> rerank() [sharpen the shortlist]
        -> match reranked text back to its source
        -> ask_claude() [generate a cited, grounded answer]
"""

def build_knowledge_base(documents, collection_name, chunk_size=150, overlap=30):
    """THE NEW IDEA. documents is a dict of {source_name: text} -- MULTIPLE
    documents, not just one. Chunks and embeds every document, tags every
    chunk's metadata with which document it came from, and stores them ALL
    in one shared collection."""
    client = chromadb.Client()
    collection = client.create_collection(name=collection_name)


    for source_name, text in documents.items():
        chunks = chunk_by_size(text, chunk_size=chunk_size, overlap=overlap)
        embeddings = [get_embedding(chunk) for chunk in chunks]
        ids = [f"{source_name}_{i}" for i in range(len(chunks))]
        metadatas = [{"source": source_name, "chunk_index": i} for i in range(len(chunks))]
        collection.add(ids=ids, documents=chunks, embeddings=embeddings, metadatas=metadatas)


    return collection




def retrieve_with_metadata(query, collection, k=6):

    too -- needed so we can trace a reranked chunk back to its source."""
    query_embedding = get_embedding(query, input_type="query")
    results = collection.query(query_embeddings=[query_embedding], n_results=k)
    return list(zip(results["documents"][0], results["metadatas"][0]))

def ask_knowledge_base(question, collection, shortlist_k=6, final_k=3):
    """The full multi-document pipeline: retrieve WITH metadata -> rerank
    -> recover each surviving chunk's source -> generate a cited answer.
    Returns the answer text AND a list of (source, chunk_text) citations."""
    shortlist = retrieve_with_metadata(question, collection, k=shortlist_k)
    shortlist_texts = [text for text, meta in shortlist]
    text_to_source = {text: meta["source"] for text, meta in shortlist}


    reranked = rerank(question, shortlist_texts, top_k=final_k)
    context_chunks = [r["document"] for r in reranked]
    citations = [(text_to_source[chunk], chunk) for chunk in context_chunks]


    context = "\n".join(f"[{source}] {chunk}" for source, chunk in citations)
    prompt = (
        f"Answer the question using ONLY the context below. Each context "
        f"line is tagged with its source in brackets -- mention which "
        f"source(s) you used in your answer. If the context doesn't "
        f"contain the answer, say so clearly.\n\n"
        f"Context:\n{context}\n\nQuestion: {question}"
    )
    answer = ask_claude(prompt, max_tokens=250)
    return answer, citations

"""
LESSON 47: MCP concepts -- servers, tools, and schemas


Welcome to Phase 7. MCP (Model Context Protocol) is an open standard,
built by Anthropic, for how an LLM-powered application discovers and
calls TOOLS -- capabilities that live OUTSIDE the model itself (reading a
file, hitting an API, querying a database, running code).


You've actually been WATCHING MCP in action this entire conversation,
without the name for it: every time I read a file, ran a Python script in
a sandbox, or searched the web to answer one of your questions, that was
me (the model) being handed a list of available TOOLS, deciding one was
needed, and the surrounding application EXECUTING that tool and handing
the result back to me. MCP is just the standardized wire format for that
exact handoff, so any MCP-compatible tool SERVER can plug into any
MCP-compatible HOST application, instead of every company inventing its
own bespoke glue code.


--- The core pieces ---
    TOOL     -- one capability, described by a NAME, a DESCRIPTION (so the
                model knows when to reach for it), and an INPUT SCHEMA
                (so the model knows exactly what arguments to provide --
                same JSON-schema idea as lesson 32's structured outputs).
    SERVER   -- a process that HOSTS one or more tools and knows how to
                actually execute them.
    HOST     -- the application (like this very conversation) that gets
                the tool list, sends it to the model alongside your
                question, and relays tool calls back and forth.


--- Before you build a real MCP server (lesson 48), see the MECHANICS ---
This lesson uses Anthropic's real "tool use" feature in the Messages API
-- the exact same tools-list + tool_use/tool_result round trip that MCP
standardizes, just with your tool functions living in THIS file instead
of a separate server process. The flow:


    1. You describe your tools (name, description, input_schema) and send
       them alongside a question.
    2. If the model decides a tool is needed, it doesn't answer directly --
       it replies with stop_reason "tool_use" and a block naming WHICH
       tool and WHAT arguments to call it with.
    3. YOUR code (the "host") actually runs that Python function.
    4. You send the result BACK to the model as a new message (same
       growing-conversation-history idea as lesson 35's chatbot), and
       the model uses it to give you a real final answer.


Lesson 48 will take these same tools and put them behind a REAL, separate
MCP server process -- but the underlying idea is exactly what you're about
to see run in this one file.

def ask_with_tools(question, tools, tool_functions, max_tokens=300):
    """Sends a question WITH a list of available tools. If the model wants
    to call one, executes it locally and sends the result back so the
    model can give a real final answer. Returns (final_answer, tool_used)."""
    messages = [{"role": "user", "content": question}]


    body = {
        "model": "claude-haiku-4-5-20251001",
        "max_tokens": max_tokens,
        "tools": tools,
        "messages": messages,
    }
    response = requests.post(anthropic_url, headers=headers, json=body)
    data = response.json()


    if data.get("stop_reason") != "tool_use":
        # The model answered directly -- no tool was needed at all.
        final_text = data["content"][0]["text"]
        return final_text, None


    # Find the tool_use block among the response's content blocks.
    tool_use_block = next(b for b in data["content"] if b["type"] == "tool_use")
    tool_name = tool_use_block["name"]
    tool_input = tool_use_block["input"]


    # THE HOST executes the tool -- the model never runs Python itself.
    tool_function = tool_functions[tool_name]
    tool_result = tool_function(**tool_input)


    # Send the result back, same growing-history idea as lesson 35.
    messages.append({"role": "assistant", "content": data["content"]})
    messages.append({
        "role": "user",
        "content": [
            {
                "type": "tool_result",
                "tool_use_id": tool_use_block["id"],
                "content": tool_result,
            }
        ],
    })


    body["messages"] = messages
    followup = requests.post(anthropic_url, headers=headers, json=body)
    followup_data = followup.json()
    final_text = next(b["text"] for b in followup_data["content"] if b["type"] == "text")


    return final_text, {"tool": tool_name, "input": tool_input, "result": tool_result}

"""
LESSON 50: An MCP tool that reads and writes real files


Every tool so far (lessons 47-49) has been pure computation -- add two
numbers, count words, look up a fake weather value. This lesson gives an
MCP tool a REAL side effect: touching the filesystem. This is the same
category of thing my own Read/Write tools do in this very conversation.


--- The new, important idea: SANDBOXING ---
A tool that can read or write ANY path on your computer is dangerous --
one wrong argument (or one confused LLM) could overwrite something
important. Real MCP file tools almost always restrict themselves to one
safe directory, and refuse anything that tries to escape it. This lesson
builds that restriction in from the start:


    1. Every read/write is confined to a single FILES_DIR folder that
       lives right next to this script.
    2. Filenames containing ".." (an attempt to climb OUT of that folder,
       e.g. "../../important_file.txt") are rejected before anything
       touches disk.


This is the same spirit as the standing project rule you already know
about -- never let a tool touch something it shouldn't be able to reach.


--- One more real thing you'll see below: how MCP reports errors ---
When a tool raises a Python exception internally, the MCP client does
NOT re-raise that exception to your calling code. Instead, you get back
a normal result object with `is_error=True` and a generic error message
in `.content`. This means you check `result.is_error` after every call
instead of wrapping everything in try/except -- more like checking an
HTTP status code than catching a crash.
"""


import os
import asyncio
from mcp.server import MCPServer
from mcp import Client


mcp = MCPServer("file-tools-demo")


FILES_DIR = os.path.join(os.path.dirname(os.path.abspath(__file__)), "mcp_files")
os.makedirs(FILES_DIR, exist_ok=True)




def safe_path(filename: str) -> str:
    """Builds a path INSIDE FILES_DIR, and refuses anything that tries to
    escape it via '..' or an absolute path."""
    if ".." in filename or os.path.isabs(filename):
        raise ValueError(f"Unsafe filename rejected: {filename}")
    return os.path.join(FILES_DIR, filename)




@mcp.tool()
def write_file(filename: str, content: str) -> str:
    """Write text content to a file inside the sandboxed files directory.


    Args:
        filename: Name of the file to write (no folders, no "..").
        content: The text content to write into the file.
    """
    path = safe_path(filename)
    with open(path, "w") as f:
        f.write(content)
    return f"Wrote {len(content)} characters to {filename}"




@mcp.tool()
def read_file(filename: str) -> str:
    """Read the text content of a file inside the sandboxed files directory.


    Args:
        filename: Name of the file to read (no folders, no "..").
    """
    path = safe_path(filename)
    if not os.path.exists(path):
        return f"No such file: {filename}"
    with open(path, "r") as f:
        return f.read()




@mcp.tool()
def list_files() -> str:
    """List every file currently in the sandboxed files directory."""
    names = os.listdir(FILES_DIR)
    if not names:
        return "No files yet."
    return ", ".join(names)


@mcp.tool()
def delete_file(filename: str) -> str:
    """Delete a file inside the sandboxed files directory.


    Args:
        filename: Name of the file to delete (no folders, no "..").
    """
    path = safe_path(filename)
    if not os.path.exists(path):
        return f"No such file to delete: {filename}"
    os.remove(path)
    return f"Deleted file: {filename}"

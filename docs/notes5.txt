"""
LESSON 36 (CAPSTONE): Text summarizer

Another combination lesson, not a new concept -- this is lesson 31's
output-format constraints (be specific about the shape you want) applied
to a genuinely useful real task: compressing a long piece of text down to
a short one.

The core trick is the SYSTEM PROMPT. "Summarize this" alone gets
inconsistent, unpredictable-length results. "Summarize this in AT MOST 3
sentences, respond with ONLY the summary, no preamble" gets something you
can actually rely on in real code -- same idea as lesson 31's "Rating:
X/10" exercise, just applied to something with real utility.

We'll also look at data["usage"] again (lesson 30) -- comparing
input_tokens (the original long text) against output_tokens (the short
summary) is a direct, numeric measure of how much you actually compressed
the text.
"""

import os
import requests
from dotenv import load_dotenv

load_dotenv(override=True)
api_key = os.getenv("ANTHROPIC_API_KEY")

url = "https://api.anthropic.com/v1/messages"

headers = {
    "x-api-key": api_key,
    "anthropic-version": "2023-06-01",
    "content-type": "application/json",
}


def summarize(text, max_sentences=3, max_tokens=200):
    """Summarizes text down to AT MOST max_sentences sentences.
    Returns (summary_text, usage_dict)."""
    system_prompt = (
        f"Summarize the following text in AT MOST {max_sentences} sentences. "
        "Respond with ONLY the summary -- no preamble, no 'Here is a summary:'."
    )
    body = {
        "model": "claude-haiku-4-5-20251001",
        "max_tokens": max_tokens,
        "system": system_prompt,
        "messages": [{"role": "user", "content": text}],
    }

    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    summary = data["content"][0]["text"]
    usage = data["usage"]
    return summary, usage

"""
LESSON 37 (CAPSTONE): Text classifier / sentiment analyzer

Final Phase 5 capstone. Combines lesson 32's structured JSON output with a
real classification task: reading a piece of text and assigning it a
clean, structured LABEL your code can actually branch on (if/elif on a
label), instead of a sentence of prose you'd have to re-read every time.

Same three techniques from lesson 32, applied to a new job:
    1. Describe the exact categories and JSON shape you want
    2. Strip markdown fences before parsing
    3. try/except around json.loads() so a bad reply never crashes you
"""

import os
import json
import requests
from dotenv import load_dotenv

load_dotenv(override=True)
api_key = os.getenv("ANTHROPIC_API_KEY")

url = "https://api.anthropic.com/v1/messages"

headers = {
    "x-api-key": api_key,
    "anthropic-version": "2023-06-01",
    "content-type": "application/json",
}


def ask(user_message, system_prompt=None, messages_history=None, max_tokens=300):
    """Same helper from lessons 31-36."""
    messages = list(messages_history) if messages_history else []
    messages.append({"role": "user", "content": user_message})

    body = {
        "model": "claude-haiku-4-5-20251001",
        "max_tokens": max_tokens,
        "messages": messages,
    }
    if system_prompt:
        body["system"] = system_prompt

    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["content"][0]["text"]


def strip_markdown_fences(text):
    """Same helper from lesson 32 -- removes ```json ... ``` fences if present."""
    text = text.strip()
    if text.startswith("```"):
        lines = text.split("\n")
        lines = lines[1:]
        if lines and lines[-1].strip() == "```":
            lines = lines[:-1]
        text = "\n".join(lines)
    return text.strip()


def classify_text(text, categories, max_tokens=150):
    """Classifies text into EXACTLY ONE of the given categories.
    Returns a dict like {"label": "...", "reason": "..."}, or None on failure."""
    categories_str = ", ".join(categories)
    system_prompt = (
        f"Classify the text into EXACTLY ONE of these categories: {categories_str}. "
        'Respond ONLY with valid JSON in this exact shape: '
        '{"label": "<one of the categories>", "reason": "<one short sentence why>"} '
        "-- no explanation outside the JSON, no markdown fences."
    )
    raw_reply = ask(text, system_prompt=system_prompt, max_tokens=max_tokens)
    cleaned = strip_markdown_fences(raw_reply)

    try:
        return json.loads(cleaned)
    except json.JSONDecodeError as e:
        print(f"Couldn't parse JSON: {e}")
        print(f"Raw reply was: {raw_reply}")
        return None

"""
LESSON 39: Embedding pipeline -- embed every chunk, and STORE it properly


Lesson 38 split a document into chunks (just strings). Lesson 33 taught you
how to turn ONE string into an embedding vector. This lesson combines them:
loop over every chunk, embed each one, and -- the actually new idea here --
STORE the text and its vector TOGETHER, not the vector alone.


--- Why store the text WITH the vector? ---
Once you have a pile of vectors, semantic search (lesson 42) works by
finding which vector is closest to a question's vector. But a vector by
itself is just 1024 meaningless-looking floats -- once you find the
CLOSEST one, you need to know what TEXT it actually came from, so you can
hand that real text to the LLM as context. If you only stored the numbers
and threw away the original chunk, "winning" the similarity search would
tell you nothing usable. So every record needs both pieces glued together:
    {"text": "the original chunk", "embedding": [0.01, -0.03, ...]}


--- Why save to a file at all? ---
Every embedding call costs a real API request (and real money, even if
tiny). Re-embedding the same documents every single time you run your
program is wasteful. Saving the finished {text, embedding} pairs to a JSON
file (lesson 20!) means you embed ONCE, then just load the file on every
future run -- this is the first hint of what a real "vector store" does,
before lessons 40-41 introduce actual vector databases.
"""


import os
import json
import time
import requests
from dotenv import load_dotenv


load_dotenv(override=True)
voyage_api_key = os.getenv("VOYAGE_API_KEY")


url = "https://api.voyageai.com/v1/embeddings"


headers = {
    "Authorization": f"Bearer {voyage_api_key}",
    "Content-Type": "application/json",
}




def get_embedding(text, input_type="document"):
    """Same helper from lesson 33/34 -- text in, a 1024-number vector out."""
    body = {
        "input": [text],
        "model": "voyage-4-lite",
        "input_type": input_type,
    }
    time.sleep(21)  # free-tier accounts are capped at 3 requests/minute
    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["data"][0]["embedding"]




def chunk_by_size(text, chunk_size, overlap=0):
    """Same helper from lesson 38."""
    if overlap >= chunk_size:
        raise ValueError("overlap must be smaller than chunk_size")
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap
    return chunks




def embed_chunks(chunks):
    """Embeds a LIST of chunks, returning a list of records that pair each
    chunk's TEXT with its EMBEDDING -- this is the actual new idea here."""
    store = []
    for chunk in chunks:
        vector = get_embedding(chunk)
        store.append({"text": chunk, "embedding": vector})
    return store




def save_store(store, filepath):
    """Saves the embedded store to a JSON file, so you don't have to
    re-embed (and re-pay for) the same chunks every time you run this."""
    with open(filepath, "w") as f:
        json.dump(store, f)




def load_store(filepath):
    """Loads a previously-saved store back from a JSON file."""
    with open(filepath, "r") as f:
        return json.load(f)

"""
LESSON 40: Real vector search with FAISS


Lesson 34 built cosine_similarity() from scratch and compared vectors one
at a time in a plain Python for loop. That works fine for 4 or 12 chunks.
It falls apart at real scale -- searching a MILLION vectors with a pure
Python loop would take way too long. FAISS (Facebook AI Similarity
Search) is a real, heavily-optimized library built specifically to search
huge piles of vectors fast. Same underlying IDEA as lesson 34 (find the
closest vector), just done by industrial-grade code instead of your own
for loop.


--- One-time setup ---
1. pip install faiss-cpu numpy
   (you already have requests/dotenv from earlier lessons)


--- The FAISS workflow, 3 steps ---
1. Build an INDEX -- tell FAISS how many dimensions your vectors have
   (1024, for Voyage embeddings), and it creates an empty searchable
   structure.
2. ADD your vectors to the index -- same {text, embedding} pairs from
   lesson 39, just handed over as one big array instead of a Python list.
3. SEARCH -- give the index a query vector and a number k (how many
   results you want), and it hands back the k closest matches, INSTANTLY,
   no matter how many vectors are stored.


One catch: FAISS only knows about NUMBERS, not your original chunk text.
Searching returns POSITIONS (indices) of the closest vectors, not the
text itself -- you use those positions to look back into your OWN store
(same {text, embedding} list from lesson 39) to get the actual readable
chunk. This is exactly why lesson 39 insisted on keeping text and
embedding paired together.
"""


import os
import time
import requests
import numpy as np
import faiss
from dotenv import load_dotenv


load_dotenv(override=True)
voyage_api_key = os.getenv("VOYAGE_API_KEY")


url = "https://api.voyageai.com/v1/embeddings"


headers = {
    "Authorization": f"Bearer {voyage_api_key}",
    "Content-Type": "application/json",
}




def get_embedding(text, input_type="document"):
    """Same helper from lessons 33/34/39."""
    body = {
        "input": [text],
        "model": "voyage-4-lite",
        "input_type": input_type,
    }
    time.sleep(21)  # free-tier accounts are capped at 3 requests/minute
    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["data"][0]["embedding"]




def chunk_by_size(text, chunk_size, overlap=0):
    """Same helper from lesson 38."""
    if overlap >= chunk_size:
        raise ValueError("overlap must be smaller than chunk_size")
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap
    return chunks




def embed_chunks(chunks):
    """Same helper from lesson 39 -- pairs each chunk's text with its vector."""
    store = []
    for chunk in chunks:
        vector = get_embedding(chunk)
        store.append({"text": chunk, "embedding": vector})
    return store




# --- Demo: build a small store, then search it for real with FAISS ---
document = (
    "Libraries have served as humanity's memory for thousands of years. "
    "The earliest known library, in Nineveh, held thousands of clay "
    "tablets recording laws, myths, and administrative records of the "
    "Assyrian empire. The Library of Alexandria later attempted something "
    "even more ambitious: collecting a copy of every known text in the "
    "ancient world, turning a single building into a hub for the world's "
    "accumulated knowledge."
)


chunks = chunk_by_size(document, chunk_size=150, overlap=30)
store = embed_chunks(chunks)
print(f"--- Store built: {len(store)} chunks ---")


# Step 1: build the index (FAISS needs to know the vector dimension)
dimension = len(store[0]["embedding"])
index = faiss.IndexFlatL2(dimension)


# Step 2: add all our vectors -- FAISS wants a numpy array of float32
vectors = np.array([record["embedding"] for record in store], dtype="float32")
index.add(vectors)
print(f"--- FAISS index built: {index.ntotal} vectors stored ---")


# Step 3: search -- embed a QUESTION, ask FAISS for the closest chunks
question = "What was the first known library?"
question_vector = np.array([get_embedding(question, input_type="query")], dtype="float32")


k = 2
distances, indices = index.search(question_vector, k)


print(f"\n--- Top {k} matches for: '{question}' ---")
for rank in range(k):
    position = indices[0][rank]
    distance = distances[0][rank]
    print(f"\nMatch {rank + 1} (distance {distance:.4f}):")
    print(store[position]["text"])

41"""
LESSON 41: Vector search with Chroma


Lesson 40 used FAISS -- fast, but low-level. YOU had to build and
maintain your own separate `store` list to remember which chunk of text
belonged to which vector (that was the whole point of lesson 39: pairing
text + embedding together yourself, since FAISS only ever hands back
positions/distances, never the text).


Chroma is a higher-level vector DATABASE. It handles that text-to-vector
pairing NATIVELY -- you hand it a document's text, a unique ID, and its
embedding all together in ONE add() call, and Chroma remembers the
connection for you. No separate list to keep in sync by hand.


--- One-time setup ---
pip install chromadb


--- The Chroma workflow, 4 steps ---
1. Create a client -- chromadb.Client() (in-memory, same idea as FAISS:
   data disappears when the program ends, unless you use
   chromadb.PersistentClient() instead).
2. Create a COLLECTION -- think of it like a table/folder for your
   vectors, given a name.
3. ADD documents -- ids, the original text, AND the embedding vector, all
   in one call. (We're using our OWN Voyage embeddings here, not
   Chroma's built-in embedding function, to stay consistent with
   everything you've already built.)
4. QUERY -- hand it a query embedding and how many results you want back.
   You get matching documents, ids, and distances, ALREADY paired up --
   Chroma did the bookkeeping lesson 39/40 made you do by hand.
"""


import os
import time
import requests
import chromadb
from dotenv import load_dotenv


load_dotenv(override=True)
voyage_api_key = os.getenv("VOYAGE_API_KEY")


url = "https://api.voyageai.com/v1/embeddings"


headers = {
    "Authorization": f"Bearer {voyage_api_key}",
    "Content-Type": "application/json",
}




def get_embedding(text, input_type="document"):
    """Same helper from lessons 33/34/39/40."""
    body = {
        "input": [text],
        "model": "voyage-4-lite",
        "input_type": input_type,
    }
    time.sleep(21)  # free-tier accounts are capped at 3 requests/minute
    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["data"][0]["embedding"]




def chunk_by_size(text, chunk_size, overlap=0):
    """Same helper from lesson 38."""
    if overlap >= chunk_size:
        raise ValueError("overlap must be smaller than chunk_size")
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunks.append(text[start:end])
        start = end - overlap
    return chunks




# --- Demo: chunk a document, embed it, store it in Chroma, search it ---
document = (
    "Libraries have served as humanity's memory for thousands of years. "
    "The earliest known library, in Nineveh, held thousands of clay "
    "tablets recording laws, myths, and administrative records of the "
    "Assyrian empire. The Library of Alexandria later attempted something "
    "even more ambitious: collecting a copy of every known text in the "
    "ancient world, turning a single building into a hub for the world's "
    "accumulated knowledge."
)


chunks = chunk_by_size(document, chunk_size=150, overlap=30)
embeddings = [get_embedding(chunk) for chunk in chunks]
ids = [f"chunk_{i}" for i in range(len(chunks))]


client = chromadb.Client()
collection = client.create_collection(name="library_docs")


collection.add(
    ids=ids,
    documents=chunks,
    embeddings=embeddings,
)
print(f"--- Added {collection.count()} documents to Chroma ---")


question = "What was the first known library?"
question_embedding = get_embedding(question, input_type="query")


results = collection.query(
    query_embeddings=[question_embedding],
    n_results=2,
)


print(f"\n--- Top 2 matches for: '{question}' ---")
for doc, distance in zip(results["documents"][0], results["distances"][0]):
    print(f"\n(distance {distance:.4f}):")
    print(doc)

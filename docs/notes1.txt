# Lesson 10: __repr__, __str__, and basic debugging


# --- Notes ---
# By default, printing an object shows something unhelpful, like:
# <__main__.Person object at 0x000001>
#
# __str__ lets you control what print() shows for an object.
# It's a method you define inside the class, and it must return a string.
#
# Example:
# class Dog:
#     def __init__(self, name):
#         self.name = name
#
#     def __str__(self):
#         return "Dog named " + self.name
#
# my_dog = Dog("Rex")
# print(my_dog)   # Dog named Rex   (instead of the ugly default)


# --- Exercise 1 ---
# Add a __str__ method to the Person class below. It should return
# a string like "Sol, age 66" (using self.name and self.age).


class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    def __str__(self):
        return "I am " + self.name + "age" + "I am " + str(self.age)

# --- Step 5: put it together ---
# 1. Create a BankAccount for yourself with a starting balance.
# 2. Deposit some money.
# 3. Try to withdraw more money than you have (should print "Insufficient funds").
# 4. Withdraw an amount you do have.
# 5. Print the account (should use your __str__).


class BankAccount:
    def __init__(self, owner, starting_balance):
        self.owner = owner
        self.balance = starting_balance


    def deposit(self, amount):
        self.balance += amount


    def withdraw(self, amount):
        if self.balance >= amount:
            self.balance -= amount
   
        else:
            print("Insufficient funds")
   
    def __str__(self):
        return "Hi, I'm with the" + self.owner + "I would like to withdraw" + "amount" + str(self.balance)


bank = BankAccount("Devins_Corp",1000000)
bank.deposit(500)
bank.withdraw(2000000)
bank.withdraw(100)
print(bank)



# Lesson 12: Error Handling (Phase 3 begins)


# --- Notes ---
# Normally, an error (exception) crashes your program. try/except lets
# you catch the error and handle it gracefully instead of crashing.
#
# Example:
# try:
#     result = 10 / 0
#     print(result)
# except ZeroDivisionError:
#     print("Can't divide by zero!")
#
# The code in "try" runs first. If it hits an error, Python jumps
# straight to "except" instead of crashing, and runs that instead.


# --- Exercise 1 ---
# Write a try/except that attempts to convert the string "abc" to an
# int using int("abc"), and catches the error with "except ValueError:",
# printing "That's not a number!" if it fails.






try:
    result = 10 / 0
    print(result)
except ZeroDivisionError:
    print("Can't divide by zero!")


try:
    int(input("Enter a number:"))
except ValueError:
    print("Thats  not a number!")


try:
    int("tuesday777:")
except ValueError:
    print("That's not a number!")


try:
    numbers = [1, 2, 3]
    print(numbers[10])
except IndexError:
    print("That index doesn't  exist!")
except ValueError:
    print("Bad value!")

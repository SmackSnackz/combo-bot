from flask import Flask, jsonify, request
import sqlite3


app = Flask(__name__)


def get_db():
    conn = sqlite3.connect("app.db")
    conn.row_factory = sqlite3.Row # lets us access columns by name, like a dict
    return conn


# ---Read all ----
@app.route("/people", methods=["GET"])
def get_people():
    conn = get_db()
    rows = conn.execute("SELECT * FROM people;").fetchall()
    conn.close()
    people = [dict(row)for row in rows]
    return jsonify(people)


# ---Read one ---
@app.route("/people/<int:person_id>", methods=["GET"])
def get_person(person_id):
    conn = get_db()
    row = conn.execute("SELECT * FROM people WHERE id = ?;",
(person_id,)).fetchone()
    conn.close()
    if row is None:
        return jsonify({"error":"person not found"}), 404
    return jsonify(dict(row))


# ---CREATE ----
@app.route("/people", methods =["POST"])
def create_person():
  data = request.get_json()
  conn= get_db()
  cursor = conn.execute(
      "INSERT INTO people (name, age, city) VALUES (?,?,?);",
      (data["name"], data["age"], data["city"]),
  )
  conn.commit()
  new_id = cursor.lastrowid
  conn.close()
  return jsonify({"id": new_id, "name": data["name"], "age": data["age"], "city": data["city"]}), 201


# ---UPDATE---
@app.route("/people/<int:person_id>", methods=["PUT"])
def update_person(person_id):
    data = request.get_json()
    conn = get_db()
    conn.execute(
      "UPDATE people SET name = ?, age = ?, city = ? WHERE id = ?;",
      (data["name"], data["age"], data["city"], person_id),
    )
    conn.commit()
    conn.close()
    return jsonify({"id": person_id, "name": data["name"], "age": data["age"], "city": data["city"]})


# --- DELETE ---
@app.route("/people/<int:person_id>", methods=["DELETE"])
def delete_person(person_id):
    conn = get_db()
    conn.execute("DELETE FROM people WHERE ID =?;",(person_id,))
    conn.commit()
    conn.close()
    return jsonify({"deleted": person_id})


if __name__ == "__main__":
    app.run(debug=True)

import requests


def get_coordinates(city_name):
    """Turn a city name into (latitude, longitude, matched_name), or None if not found."""
    url =  "https://geocoding-api.open-meteo.com/v1/search"
    params = {"name": city_name, "count": 1}


    response = requests.get(url,params=params)


    if response.status_code!= 200:
        print(f"Geocoding requests failed (status {response.status_code})")
        return None


    data = response.json()


    if "results" not in data or len(data["results"]) == 0:
      print(f"Couldn't find a place called '{city_name}'")
      return None


    match = data["results"][0]
    return match["latitude"], match["longitude"], match["name"]      


def get_weather(lat,lon):
    """Get the current weather for a given latitude/longitude."""
    url = "https://api.open-meteo.com/v1/forecast"
    params = {"latitude": lat, "longitude": lon, "current_weather": True}


    response = requests.get(url, params=params)




    if response.status_code != 200:
        print(f"Weather request failed (status {response.status_code})")
        return None


    data = response.json()
    return data["current_weather"]


def show_weather_for(city_name):
    coords = get_coordinates(city_name)
    if coords is None:
       return # get_coorddinates already printed why
 
    lat, lon, matched_name = coords
    weather = get_weather(lat, lon)
    if weather is None:
        return


    print(f"\n--- Weather for {matched_name}---")
    print(f"Temperature: {weather['temperature']} C")
    print(f"Wind speed: {weather['windspeed']} km/h")

"""
LESSON 29: What an LLM actually is, and what "tokenization" means


Welcome to Phase 5 -- this is where the AI/ML side of your goal actually
starts. Everything in Phases 1-4 (Python, OOP, SQL, APIs, Flask) was the
plumbing. Starting now, we build on top of it.


--- What an LLM (Large Language Model) actually is ---
At the core, an LLM is a program that has been trained on huge amounts
of text to do ONE thing extremely well: given some text so far, predict
what text is most likely to come next. That's it. Everything else --
answering questions, writing code, holding a conversation -- comes from
that one skill applied over and over, one small piece at a time.


It does NOT "look up" answers in a database (that's what you built in
Phase 4 -- a totally different thing). It generates its answer piece by
piece based on patterns learned during training.


--- What "tokenization" means ---
An LLM can't read letters or words directly -- it only understands
numbers. Tokenization is the step that converts text into numbers:


    1. Split the text into small pieces called TOKENS. A token is often
       a whole word, but can also be a piece of a word, or a single
       punctuation mark. ("unbelievable" might become "un" + "believable")
    2. Each unique token gets assigned a number (its token ID) using a
       lookup table called a VOCABULARY -- basically a big dict, exactly
       like the dicts you've been using since lesson 2.
    3. The model receives a list of numbers, does its prediction, and
       spits out more numbers, which get converted BACK to text using
       that same vocabulary in reverse.


Real tokenizers (used by GPT, Claude, etc.) are much smarter than what
we're building below -- they're trained on huge amounts of text to find
the most efficient way to split words. But the CONCEPT is identical to
what you're about to build by hand: text -> tokens -> numbers -> (later)
numbers -> tokens -> text.
"""


# --- A tiny tokenizer, built from scratch with plain Python ---
# This is NOT how real tokenizers work internally, but the concept --
# build a vocabulary, encode text to numbers, decode numbers back to
# text -- is exactly the same idea.


import re


def simple_tokenize(text):
    """Splits text into words and punctuation as separate tokens."""
    # \w+ matches a run of letters/numbers (a word), [^\w\s] matches a
    # single punctuation character. This is just pattern matching --
    # you don't need to understand regex deeply to get the concept here.
    return re.findall(r"\w+|[^\w\s]", text.lower())


def build_vocabulary(tokens):
    """Assigns each UNIQUE token a number, in the order first seen."""
    vocabulary = {}
    for token in tokens:
        if token not in vocabulary:
            vocabulary[token] = len(vocabulary)
    return vocabulary


def encode(text, vocabulary):
    """Text -> list of token IDs, using an existing vocabulary."""
    tokens = simple_tokenize(text)
    return [vocabulary[token] for token in tokens]


def decode(token_ids, vocabulary):
    """List of token IDs -> text, using the SAME vocabulary in reverse."""
    # We need the reverse mapping: number -> token (instead of token -> number)
    id_to_token = {token_id: token for token, token_id in vocabulary.items()}
    return " ".join(id_to_token[token_id] for token_id in token_ids)


# --- Try it out ---
sample_text = "The cat sat on the mat. The cat was happy!"


tokens = simple_tokenize(sample_text)
print("--- Tokens ---")
print(tokens)


vocabulary = build_vocabulary(tokens)
print("\n--- Vocabulary (token -> id) ---")
print(vocabulary)


encoded = encode(sample_text, vocabulary)
print("\n--- Encoded (text -> numbers) ---")
print(encoded)


decoded = decode(encoded, vocabulary)
print("\n--- Decoded (numbers -> text) ---")
print(decoded)


print(f"\nUnique tokens in vocabulary: {len(vocabulary)}")
print(f"Total tokens in the sentence: {len(tokens)}")
print("Notice 'the' and 'cat' only appear ONCE in the vocabulary, even")
print("though they appear TWICE in the sentence -- that's the whole point")
print("of a vocabulary: reusing the same number for the same token.")

"""
LESSON 31: Prompt engineering -- system prompts, few-shot examples, output format


You already know how to CALL an LLM (lesson 30). Prompt engineering is about
controlling WHAT COMES BACK, using nothing but the words you send. No new
libraries, no new API shape -- just three techniques layered onto the same
requests.post() call you already know.


--- 1. System prompts ---
So far every message you've sent has had role "user". Anthropic's API also
accepts a separate top-level "system" field (NOT inside the messages list) --
this sets the model's persona/behavior/rules for the ENTIRE conversation,
before it sees anything you ask. Same request, different system prompt,
wildly different answer to the exact same question.


--- 2. Few-shot examples ---
"Few-shot" means: instead of just describing what you want, you SHOW the
model 2-3 example input/output pairs first (as fake user/assistant messages),
then ask your real question last. The model pattern-matches the FORMAT of
your examples onto your new input. This is often more reliable than
describing the format in words.


--- 3. Output formatting ---
You can just... ask for the shape you want. "Respond with ONLY the number,
nothing else." "Respond in exactly this format: Rating: X/10." The model
usually complies, though not with 100% guaranteed reliability -- lesson 32
covers how to make this more bulletproof.
"""


import os
import requests
from dotenv import load_dotenv


load_dotenv(override=True)
api_key = os.getenv("ANTHROPIC_API_KEY")




url = "https://api.anthropic.com/v1/messages"






headers = {
    "x-api-key":api_key,
    "anthropic-version": "2023-06-01",
    "content-type": "application/json",
}


def ask(user_message, system_prompt=None, messages_history=None, max_tokens=150):
    """Small helper so we don't retype the request boilerplate everytime.
    system_prompt is optional. messages_history lets us pass few_shot
      examples
    (a list of {"role":..., "content":...} dicts) BEFORE the real question."""
    messages = list(messages_history) if messages_history else[]
    messages.append({"role": "user", "content": user_message})




    body = {
        "model": "claude-haiku-4-5-20251001",
        "max_tokens": max_tokens,
        "messages": messages,
    }
    if system_prompt:
        body["system"] = system_prompt




    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["content"][0]["text"]

"""
LESSON 32: Structured outputs -- getting reliable JSON back from an LLM


Lesson 31 showed the model won't always follow formatting instructions
exactly -- it can refuse, add commentary, or drift. For most real
applications (APIs, automations, saving to a database) "usually good text"
isn't good enough. You need the reply as DATA -- a Python dict, via
json.loads() -- not just a sentence you eyeball and trust.


Three techniques, used together:


--- 1. Ask for JSON explicitly, with an exact schema ---
Describe the exact keys and types you want in the system prompt. Vague
("give me some JSON") gets vague results. Specific field names, types, and
an example shape gets specific results.


--- 2. Strip markdown fences before parsing ---
Models are trained on a lot of markdown, so they often wrap JSON in
```json ... ``` fences even when told not to. Real code has to handle
this: strip the fences before handing the text to json.loads().


--- 3. Always wrap json.loads() in try/except ---
No matter how good your prompt is, the model can still return something
that isn't valid JSON (extra commentary, a typo, a trailing comma). Never
trust it blindly -- catch json.JSONDecodeError and handle the failure
instead of letting your program crash.
"""


import os
import json
import requests
from dotenv import load_dotenv


load_dotenv(override=True)
api_key = os.getenv("ANTHROPIC_API_KEY")


url = "https://api.anthropic.com/v1/messages"


headers = {
    "x-api-key": api_key,
    "anthropic-version": "2023-06-01",
    "content-type": "application/json",
}




def ask(user_message, system_prompt=None, messages_history=None, max_tokens=300):
    """Same helper from lesson 31 -- sends one request, returns the reply text."""
    messages = list(messages_history) if messages_history else []
    messages.append({"role": "user", "content": user_message})


    body = {
        "model": "claude-haiku-4-5-20251001",
        "max_tokens": max_tokens,
        "messages": messages,
    }
    if system_prompt:
        body["system"] = system_prompt


    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["content"][0]["text"]




def strip_markdown_fences(text):
    """Models often wrap JSON in ```json ... ``` even when told not to.
    Remove the fences (if present) so json.loads() gets a clean string."""
    text = text.strip()
    if text.startswith("```"):
        lines = text.split("\n")
        lines = lines[1:]          # drop the opening ``` (or ```json) line
        if lines and lines[-1].strip() == "```":
            lines = lines[:-1]     # drop the closing ``` line
        text = "\n".join(lines)
    return text.strip()




def ask_for_json(user_message, schema_description, max_tokens=300):
    """Asks the model for JSON matching a described schema, and safely
    parses the reply. Returns a Python dict/list on success, or None on
    failure (printing why)."""
    system_prompt = (
        "You respond ONLY with valid JSON matching this schema, and "
        f"nothing else -- no explanation, no markdown fences: {schema_description}"
    )
    raw_reply = ask(user_message, system_prompt=system_prompt, max_tokens=max_tokens)
    cleaned = strip_markdown_fences(raw_reply)


    try:
        return json.loads(cleaned)
    except json.JSONDecodeError as e:
        print(f"Couldn't parse JSON: {e}")
        print(f"Raw reply was: {raw_reply}")
        return None

"""
LESSON 33: Embeddings -- turning text into "meaning vectors"


--- What an embedding actually is ---
In lesson 29 you built a TOKENIZER: text -> arbitrary lookup numbers (token
IDs), just so the model could read the text at all. Those numbers carry NO
meaning by themselves -- token 47 isn't "more similar" to token 48 just
because they're next to each other. It's just a lookup table.


An EMBEDDING is a completely different kind of number-conversion. It turns
a whole piece of text into a long list of numbers (a VECTOR -- often 1024
numbers long) that represents its MEANING. The key property: texts with
SIMILAR meaning produce vectors that are mathematically close together,
and texts with DIFFERENT meaning produce vectors that are far apart. "The
cat sat on the mat" and "A feline rested on the rug" would produce very
similar vectors, even though they don't share almost any of the same
words -- because an embedding model was trained specifically to capture
meaning, not just word-matching.

MEANING, not exact keyword match) and RAG (Phase 6) -- you'll embed a
question and embed a pile of documents, then find which documents have

"""
LESSON 34: Vector math -- proving which embeddings are actually "close"


Lesson 33 kept saying "trust me, A and B are closer together than either is
to C" without ever proving it. This lesson builds the actual math, from
scratch, in plain Python -- no libraries, no imports beyond what you
already know. Three small functions, each one plain arithmetic:


--- 1. Dot product ---
Pair up each number in vector A with the matching number in vector B
(position 1 with position 1, position 2 with position 2, ...), multiply
each pair, then add up all the results into one total. That's it --
multiply, then sum.


--- 2. Magnitude (length) of a vector ---
Square every number in the vector, add them all up, then take the square
root of that total. This is the Pythagorean theorem (a^2 + b^2 = c^2)
stretched from 2 dimensions to however many the vector has (1024, in our
case).


--- 3. Cosine similarity ---
    cosine_similarity(A, B) = dot_product(A, B) / (magnitude(A) * magnitude(B))


   
The division by both magnitudes matters -- it "normalizes" the result so
it only reflects DIRECTION (meaning), not how large the raw numbers happen
to be. The result always lands between -1 and 1:
    1.0  -> pointing in the exact same direction (as similar as possible)
    0.0  -> unrelated / no meaningful relationship
   -1.0  -> pointing in opposite directions


This one calculation is the entire foundation of semantic search and RAG
(Phase 6) -- embed a question, embed a pile of documents, compute cosine
similarity between the question and every document, and whichever
document scores highest is the most relevant one.
"""


import os
import time
import requests
from dotenv import load_dotenv


load_dotenv(override=True)
voyage_api_key = os.getenv("VOYAGE_API_KEY")


url = "https://api.voyageai.com/v1/embeddings"


headers = {
    "Authorization": f"Bearer {voyage_api_key}",
    "Content-Type": "application/json",
}




def get_embedding(text, input_type="document"):
    """Same helper from lesson 33 -- text in, a 1024-number vector out."""
    body = {
        "input": [text],
        "model": "voyage-4-lite",
        "input_type": input_type,
    }
    time.sleep(21)  # free-tier accounts are capped at 3 requests/minute
    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["data"][0]["embedding"]




def dot_product(vec_a, vec_b):
    """Multiply matching positions together, sum the results."""
    total = 0
    for a, b in zip(vec_a, vec_b):
        total += a * b
    return total




def magnitude(vec):
    """Length of a vector -- Pythagorean theorem stretched to N dimensions."""
    total = 0
    for x in vec:
        total += x ** 2
    return total ** 0.5




def cosine_similarity(vec_a, vec_b):
    """Returns a number from -1 to 1. Closer to 1 means more similar meaning."""
    return dot_product(vec_a, vec_b) / (magnitude(vec_a) * magnitude(vec_b))

The one core idea, restated from lesson 30's exercise 4: the model has NO
memory of its own. Every single call, you resend the ENTIRE conversation
so far. "Remembering" isn't something the model does -- it's something
YOUR CODE does, by keeping a growing list and handing the whole thing back
every time.


The pattern, every turn:
    1. Get the user's new message with input()
    2. Call ask(), passing the FULL conversation_history so far
    3. Append the user's message AND the model's reply to
       conversation_history -- so NEXT turn, the model can "see" this
       exchange too
    4. Print the reply, loop back to step 1
"""


import os
import requests
from dotenv import load_dotenv


load_dotenv(override=True)
api_key = os.getenv("ANTHROPIC_API_KEY")


url = "https://api.anthropic.com/v1/messages"


headers = {
    "x-api-key": api_key,
    "anthropic-version": "2023-06-01",
    "content-type": "application/json",
}




def ask(user_message, system_prompt=None, messages_history=None, max_tokens=300):
    """Same helper from lessons 31-32 -- sends one request, returns reply text."""
    messages = list(messages_history) if messages_history else []
    messages.append({"role": "user", "content": user_message})


    body = {
        "model": "claude-haiku-4-5-20251001",
        "max_tokens": max_tokens,
        "messages": messages,
    }
    if system_prompt:
        body["system"] = system_prompt


    response = requests.post(url, headers=headers, json=body)
    data = response.json()
    return data["content"][0]["text"]




# --- Demo: two hardcoded turns, proving history actually grows and gets used ---
conversation_history = []


# Turn 1
user_msg_1 = "My name is Roy."
reply_1 = ask(user_msg_1, messages_history=conversation_history)
print(f"You: {user_msg_1}")
print(f"Bot: {reply_1}")


conversation_history.append({"role": "user", "content": user_msg_1})
conversation_history.append({"role": "assistant", "content": reply_1})


# Turn 2 -- does the model remember, using ONLY conversation_history as proof?
user_msg_2 = "What's my name?"
reply_2 = ask(user_msg_2, messages_history=conversation_history)
print(f"\nYou: {user_msg_2}")
print(f"Bot: {reply_2}")


conversation_history.append({"role": "user", "content": user_msg_2})
conversation_history.append({"role": "assistant", "content": reply_2})


print(f"\n(conversation_history now holds {len(conversation_history)} messages)")
